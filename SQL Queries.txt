     FROM dim_customer

Request 2 :
      WITH CTE1 AS (
	SELECT COUNT(DISTINCT product_code) AS 
           unique_products_2020
      FROM fact_sales_monthly
      WHERE fiscal_year = 2020
      ) , 
      CTE2 AS (
	SELECT COUNT(DISTINCT product_code) AS 
           unique_products_2021
      FROM fact_sales_monthly
      WHERE fiscal_year = 2021
     )
     SELECT 
       unique_products_2020 ,
       unique_products_2021 , 
       ROUND((unique_products_2021 - unique_products_2020)/unique_products_2020*100,2) AS percentage_chg
     FROM CTE1,CTE2;

Request 3 :
      SELECT segment , 
             COUNT(DISTINCT product_code) AS product_count 
      FROM dim_product 
      GROUP BY segment 
      ORDER BY product_count DESC;

Request 4 :
     WITH CTE1 AS (
       SELECT segment , 
	      COUNT(DISTINCT d.product_code) AS product_count_2020
       FROM dim_product d 
       LEFT JOIN fact_sales_monthly f 
       ON d.product_code = f.product_code
       WHERE fiscal_year = 2020 
       GROUP BY segment
      ),
      CTE2 AS (
        SELECT segment , 
	   COUNT(DISTINCT d.product_code) AS product_count_2021
        FROM dim_product d 
        LEFT JOIN fact_sales_monthly f 
        ON d.product_code = f.product_code
        WHERE fiscal_year = 2021
        GROUP BY segment
      )
      SELECT c1.segment , 
       product_count_2020,
       product_count_2021, 
       (product_count_2021 - product_count_2020) AS difference
      FROM CTE1 c1
      JOIN CTE2 c2
      ON c1.segment = c2.segment
      ORDER BY difference DESC;

Request 5 :
      SELECT d.product_code ,
       product ,
       manufacturing_cost
      FROM dim_product d
      LEFT JOIN fact_manufacturing_cost f
      ON d.product_code = f.product_code
      WHERE manufacturing_cost = (
         SELECT MAX(manufacturing_cost) AS manufacturing_cost
         FROM fact_manufacturing_cost
      )
     UNION 
     SELECT d.product_code , 
	    product , 
            manufacturing_cost
     FROM dim_product d 
     LEFT JOIN fact_manufacturing_cost f 
     ON d.product_code = f.product_code
     WHERE manufacturing_cost = (
		SELECT MIN(manufacturing_cost) AS manufacturing_cost 
		FROM fact_manufacturing_cost
        )


Request 6:
      SELECT f.customer_code ,
       customer ,
       ROUND(AVG(pre_invoice_discount_pct)*100,2) AS average_discount_percentage
      FROM fact_pre_invoice_deductions f 
      INNER JOIN dim_customer d
       ON f.customer_code = d.customer_code
      WHERE fiscal_year = 2021 
       AND market = 'India'
      GROUP BY f.customer_code ,
         customer
      ORDER BY average_discount_percentage DESC 
      LIMIT 5;

Request 7 :
      SELECT monthname(date) AS Month , 
       YEAR(date) AS Year , 
       ROUND(SUM(gross_price * sold_quantity)/1000000,2) AS Gross_sales_Amount_in_million
      FROM fact_gross_price f1
      INNER JOIN fact_sales_monthly f2 
       ON f1.product_code = f2.product_code 
       AND f1.fiscal_year = f2.fiscal_year
      INNER JOIN dim_customer d 
       ON f2.customer_code = d.customer_code
      WHERE customer = 'Atliq Exclusive'
      GROUP BY monthname(date) , YEAR(date);

Request 8 :
       SELECT CASE 
        WHEN MONTH(date) IN (9,10,11) THEN 'Q1'
        WHEN MONTH(date) IN (12,1,2) THEN 'Q2'
        WHEN MONTH(date) IN (3,4,5) THEN 'Q3'
        WHEN MONTH(date) IN (6,7,8) THEN 'Q4'
        END AS Quarter ,
        ROUND(SUM(sold_quantity)/1000000,2) AS total_sold_quantity_in_millions
       FROM fact_sales_monthly
       WHERE fiscal_year = 2020
       GROUP BY Quarter
       ORDER BY total_sold_quantity_in_millions DESC;

Request 9 :
       SELECT channel , 
           ROUND(SUM(gross_price*sold_quantity)/1000000,2) AS gross_sales_mln ,
	   ROUND(SUM(gross_price*sold_quantity) / SUM(SUM(gross_price*sold_quantity)) OVER()*100,2) AS percentage
       FROM fact_gross_price f1
       INNER JOIN fact_sales_monthly f2 
        ON f1.product_code = f2.product_code 
         AND f1.fiscal_year = f2.fiscal_year 
       INNER JOIN dim_customer d 
        ON f2.customer_code = d.customer_code
       WHERE f1.fiscal_year = 2021
       GROUP BY channel 
       ORDER BY percentage DESC;  

Request 10 :
     WITH DenseRank AS (
      SELECT division ,
      d.product_code ,
      product,
      SUM(sold_quantity) AS total_sold_quantity,
      DENSE_RANK() OVER(partition by division ORDER BY SUM(sold_quantity) DESC) AS rank_order
      FROM dim_product d
      JOIN fact_sales_monthly f
       ON d.product_code = f.product_code
      WHERE fiscal_year = 2021
      GROUP BY division , d.product_code , product
      )
      SELECT *
      FROM DenseRank
      WHERE rank_order <=3;
